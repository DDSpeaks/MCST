// MCST Tracker Bridge 1.0 Host
// Internal bridge build: V157. Bridge protocol: V2.
// Add this indicator to one chart in each MultiCharts64 instance that owns
// the Order and Position Tracker used by MCST-Watchdog.

External:
    "C:\MCExtras\MCST-TrackerBridge.dll",
    Int, "MCBridge_Initialize";

External:
    "C:\MCExtras\MCST-TrackerBridge.dll",
    Int, "MCBridge_Heartbeat";

External:
    "C:\MCExtras\MCST-TrackerBridge.dll",
    Int, "MCBridge_GetState";

Variables:
    BridgeStarted(False),
    BridgeResult(0),
    BridgeState(0);

if not BridgeStarted then begin
    BridgeResult = MCBridge_Initialize();
    BridgeStarted = True;
end;

if LastBarOnChart then begin
    BridgeResult = MCBridge_Heartbeat();
    BridgeState = MCBridge_GetState();
end;
